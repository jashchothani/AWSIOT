Wire the DHT11

Connect the sensor before writing any AWS code.

3-pin DHT11 module

DHT11 VCC   ─────────  NodeMCU 3V3
DHT11 DATA  ─────────  NodeMCU D4
DHT11 GND   ─────────  NodeMCU GND
⚠️
Bare 4-pin DHT11?

Use approximately a 10kΩ pull-up resistor between DATA and VCC.

Test the DHT11

Always verify the hardware before introducing AWS.

✓
Important debugging rule

If the sensor does not work locally, AWS will not fix it. Test one layer at a time. 

#include <DHT.h>

#define DHTPIN D4
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);

void setup() {

  Serial.begin(115200);

  dht.begin();

  Serial.println("\nDHT11 Test");

}

void loop() {

  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();

  if (isnan(temperature) || isnan(humidity)) {

    Serial.println("DHT11 read failed!");

    delay(2000);

    return;
  }

  Serial.printf(
    "Temperature: %.1f C\n",
    temperature
  );

  Serial.printf(
    "Humidity: %.1f %%\n",
    humidity
  );

  Serial.println("----------------");

  delay(2000);
}

Example policy for initial testing: 

{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": "iot:Connect",
      "Resource": "*"
    },
    {
      "Effect": "Allow",
      "Action": "iot:Publish",
      "Resource": "*"
    },
    {
      "Effect": "Allow",
      "Action": "iot:Subscribe",
      "Resource": "*"
    },
    {
      "Effect": "Allow",
      "Action": "iot:Receive",
      "Resource": "*"
    }
  ]
}
🔐
Security note

This wildcard policy is suitable for a first test. For production, restrict resources to the required device and MQTT topics.

LittleFS certificate storage

Store certificates separately from your Arduino sketch.

Why LittleFS?

LittleFS stores files inside the ESP8266 flash filesystem. Your certificate files can therefore be loaded at runtime instead of embedding their contents directly into the sketch.

Q — Check Arduino IDE

Help → About Arduino IDE → Arduino IDE 2.2.1 or newer

R — Install uploader

  1. Close Arduino IDE.
  2. Download the latest LittleFS uploader .vsix.
  3. Create: C:\Users\YOUR_USERNAME\.arduinoIDE\plugins\
  4. Place the VSIX file inside that folder.
  5. Reopen Arduino IDE.

S — Verify uploader

> Ctrl + Shift + P
> Upload LittleFS

The final ESP8266 sketch

Connect Wi-Fi, load certificates, establish TLS, read the DHT11 and publish MQTT data.

✏️
Only change your own values

Replace the Wi-Fi credentials and AWS endpoint. Keep certificate filenames and topic consistent. 

#include <ESP8266WiFi.h>
#include <LittleFS.h>
#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <DHT.h>
#include <time.h>


// ==============================
// WiFi & AWS
// ==============================

const char* WIFI_SSID =
    "YOUR_WIFI_NAME";

const char* WIFI_PASSWORD =
    "YOUR_WIFI_PASSWORD";

const char* AWS_ENDPOINT =
    "YOUR-ENDPOINT-ats.iot.ap-south-1.amazonaws.com";

const char* MQTT_TOPIC =
    "home/esp8266/dht11";


// ==============================
// DHT11
// ==============================

#define DHTPIN D4
#define DHTTYPE DHT11

DHT dht(
    DHTPIN,
    DHTTYPE
);


// ==============================
// Secure MQTT
// ==============================

WiFiClientSecure secureClient;

PubSubClient mqttClient(
    secureClient
);

BearSSL::X509List* rootCA = nullptr;

BearSSL::X509List* deviceCert = nullptr;

BearSSL::PrivateKey* privateKey = nullptr;


// ==============================
// Load certificates
// ==============================

bool loadCertificates() {

  Serial.println(
      "\nLoading AWS certificates..."
  );


  File rootFile =
      LittleFS.open(
          "/AmazonRootCA1.pem",
          "r"
      );


  if (!rootFile) {

    Serial.println(
        "ERROR: AmazonRootCA1.pem not found!"
    );

    return false;
  }


  rootCA =
      new BearSSL::X509List(
          rootFile,
          rootFile.size()
      );

  rootFile.close();


  File certFile =
      LittleFS.open(
          "/device.crt",
          "r"
      );


  if (!certFile) {

    Serial.println(
        "ERROR: device.crt not found!"
    );

    return false;
  }


  deviceCert =
      new BearSSL::X509List(
          certFile,
          certFile.size()
      );

  certFile.close();


  File keyFile =
      LittleFS.open(
          "/private.key",
          "r"
      );


  if (!keyFile) {

    Serial.println(
        "ERROR: private.key not found!"
    );

    return false;
  }


  privateKey =
      new BearSSL::PrivateKey(
          keyFile,
          keyFile.size()
      );

  keyFile.close();


  Serial.println(
      "AWS certificates loaded successfully!"
  );


  return true;
}


// ==============================
// WiFi
// ==============================

void connectWiFi() {

  Serial.print(
      "\nConnecting to WiFi"
  );


  WiFi.mode(WIFI_STA);

  WiFi.begin(
      WIFI_SSID,
      WIFI_PASSWORD
  );


  while (
      WiFi.status() != WL_CONNECTED
  ) {

    delay(500);

    Serial.print(".");
  }


  Serial.printf(
      "\nWiFi connected! IP: %s\n",
      WiFi.localIP()
          .toString()
          .c_str()
  );
}


// ==============================
// NTP time
// ==============================

void setClock() {

  Serial.println(
      "\nSynchronizing time..."
  );


  configTime(
      0,
      0,
      "pool.ntp.org",
      "time.nist.gov"
  );


  time_t now =
      time(nullptr);


  int attempts = 0;


  while (
      now < 8 * 3600 * 2 &&
      attempts < 30
  ) {

    delay(500);

    Serial.print(".");

    now = time(nullptr);

    attempts++;
  }


  if (
      now >= 8 * 3600 * 2
  ) {

    Serial.println(
        "\nTime synchronized!"
    );

  } else {

    Serial.println(
        "\nWARNING: Time sync failed."
    );
  }
}


// ==============================
// AWS connection
// ==============================

void connectAWS() {

  while (
      !mqttClient.connected()
  ) {

    Serial.print(
        "\nConnecting to AWS IoT..."
    );


    String clientID =
        "ESP8266-DHT11-" +
        String(
            ESP.getChipId()
        );


    if (
        mqttClient.connect(
            clientID.c_str()
        )
    ) {

      Serial.println(
          "CONNECTED!"
      );

    } else {

      Serial.printf(
          "FAILED, MQTT state = %d\n",
          mqttClient.state()
      );

      delay(5000);
    }
  }
}


// ==============================
// Setup
// ==============================

void setup() {

  Serial.begin(
      115200
  );

  delay(1000);


  Serial.println(
      "\n================================"
  );

  Serial.println(
      "ESP8266 + DHT11 + AWS IoT Core"
  );

  Serial.println(
      "================================"
  );


  dht.begin();


  if (
      !LittleFS.begin()
  ) {

    Serial.println(
        "ERROR: LittleFS failed!"
    );

    return;
  }


  if (
      !loadCertificates()
  ) {

    Serial.println(
        "Certificate loading failed!"
    );

    return;
  }


  connectWiFi();

  setClock();


  secureClient.setTrustAnchors(
      rootCA
  );


  secureClient.setClientRSACert(
      deviceCert,
      privateKey
  );


  mqttClient.setServer(
      AWS_ENDPOINT,
      8883
  );


  connectAWS();
}


// ==============================
// Main loop
// ==============================

void loop() {

  if (
      !mqttClient.connected()
  ) {

    connectAWS();
  }


  mqttClient.loop();


  float temperature =
      dht.readTemperature();


  float humidity =
      dht.readHumidity();


  if (
      isnan(temperature) ||
      isnan(humidity)
  ) {

    Serial.println(
        "DHT11 reading failed!"
    );

    delay(2000);

    return;
  }


  String payload =
      "{\"device\":\"ESP8266-DHT11\","
      "\"temperature\":" +
      String(
          temperature,
          1
      ) +
      ",\"humidity\":" +
      String(
          humidity,
          1
      ) +
      "}";


  Serial.printf(
      "\nTemperature: %.1f C | Humidity: %.1f %%\n",
      temperature,
      humidity
  );


  Serial.println(
      "MQTT Payload: " +
      payload
  );


  bool success =
      mqttClient.publish(
          MQTT_TOPIC,
          payload.c_str()
      );


  Serial.println(
      success
      ? "AWS: Message published!"
      : "AWS: Publish FAILED!"
  );


  delay(5000);
}

Troubleshooting

Find the layer that failed instead of changing everything.

Error Check
LittleFS failed! Board selection and LittleFS upload.
AmazonRootCA1.pem not found Check the data/ folder and upload LittleFS again.
device.crt not found Verify the exact filename.
private.key not found Verify the key exists inside LittleFS.
Wi-Fi won't connect Check credentials and use a 2.4 GHz network.
MQTT state = -2 Check endpoint, Wi-Fi, certificates and time.
MQTT state = 5 Check certificate status and IoT policy.
⏱️
Don't forget NTP time

TLS certificate validation depends on the device having a correct clock.
